## 6. 验收命令

```bash
uv lock            # -> 0，生成 uv.lock
uv sync --locked   # -> 0，安装锁定环境
.venv/bin/python -c "import torch, gymnasium, stable_baselines3, scipy, pandapower; print('imports ok')"  # -> 0
# HiGHS 二元互斥微例（可直接复制执行）：二元变量 x1,x2 ∈ {0,1}，互斥约束 x1+x2<=1，
# 期望最优解 x=[1,0]，目标值 1.0（最大化 x1 + 0.5*x2）。
.venv/bin/python - <<'PY'
import numpy as np
from scipy.optimize import milp, LinearConstraint, Bounds
c = -np.array([1.0, 0.5])                       # minimize -c·x  <=>  maximize x1 + 0.5*x2
constraints = LinearConstraint(np.array([[1.0, 1.0]]), -np.inf, 1.0)  # x1 + x2 <= 1
res = milp(c=c, constraints=constraints, integrality=np.ones(2), bounds=Bounds(0.0, 1.0))
assert res.success, res.message
x = np.round(res.x, 6)
print("solver status :", res.success, "|", res.message)
print("solution x    :", x.tolist())
print("objective(max):", float(-res.fun))
assert x[0] == 1.0 and x[1] == 0.0, f"unexpected {x}"
assert abs(float(-res.fun) - 1.0) < 1e-9
print("PASS: expected x=[1,0], objective=1.0")
PY
.venv/bin/python -c "import torch; print('mps', torch.backends.mps.is_available(), 'cuda', torch.cuda.is_available())"  # -> 0，cuda=False
git status --short   # -> 空
git diff --check     # -> 空
git diff --exit-code paper-baseline -- environment.yml   # -> 0（environment.yml 未改）
```
